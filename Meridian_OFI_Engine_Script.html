// Meridian Engine v2.2 — unique SMC + VWAP synthesis + Order Flow Intelligence
// Pillars: Regime Engine · Liquidity Memory Map · Event Lattice · 3-Gate Core
//          Path Integrity · Dynamic Risk Geometry · Session DNA · OFI (CVD + Stack)

// v2.2 STACKED IMBALANCE (built on v2.1):
//  - Native request.footprint() diagonal buy/sell imbalance when available
//  - Consecutive same-side rows (≥ minStack) → lattice event type 10
//  - Stack zones optionally registered as high-intensity magnets
//  - CVD multi-bar proxy fallback when footprint data is na

//  - Cumulative Volume Delta (session/daily/continuous reset)
//  - Absorption & imbalance detection at SMC / VWAP / magnet levels
//  - New lattice event types 8 (absorption) and 9 (imbalance)
//  - Direction-aware density feed into existing grade → Gate 3 pipeline
//  - Minimal visuals + dashboard row; path-integrity consumes the same evidence
//
// v2 CHANGES FROM v1 (bug fixes, not feature additions):
//  1. Lattice grade's magnet boost is now direction-matched (was direction-blind:
//     any nearby magnet used to boost BOTH long and short grades equally).
//  2. SL is now adjusted to match the ATR-floored risk figure used for TP math
//     (was: SL could sit closer than the risk the TP1/TP2/TP3 targets assumed).
//  3. Entry signals (and their alerts) now require barstate.isconfirmed, so they
//     can no longer fire mid-bar off a live price and vanish before the close.
//  4. VWAP "Session" anchor / RTH gate no longer hardcoded to NY 09:30-16:00 for
//     24/7 assets — crypto symbols get a full-day window instead of losing VWAP,
//     bands, and band-confluence gating for most of the day.
//  5. Event lattice now counts each event TYPE once per window (bitmask dedup)
//     instead of raw hits, so a single displacement move that fires sweep+CHoCH+
//     BOS+VWAP-reclaim in the same 2-3 bars can't also inflate the score further
//     via repeats of the same event type.

f_nearestOpposingMagnet(int dir, float fromPx) =>

                    float d = math.abs(p.price - fromPx)
                    if d < bestDist
                        bestDist := d
                        best := p.price
    best

// FIX: density used to count every raw event hit. A single displacement move commonly
// fires sweep + CHoCH + BOS + VWAP-reclaim within 2-3 bars at nearly the same price —
// those are correlated tags of ONE move, not four independent confirmations, and worse,
// two sweeps or two band-touches in the window used to double-count the same evidence
// type. Now each event TYPE (see codes above f_pushEvent) counts at most once per
// window via a bitmask, so density reflects how many DIFFERENT kinds of confluence
// showed up, not how many times any one kind repeated.
f_latticeDensity(int dir) =>
    int mask = 0
    float prox = latticeATR * atr
    int n = array.size(evBars)
    if n > 0
        for i = 0 to n - 1
            int eb = array.get(evBars, i)
            int ed = array.get(evDirs, i)
            float ep = array.get(evPrices, i)
            int et = array.get(evTypes, i)
            if ed == dir and (bar_index - eb) <= latBars and math.abs(ep - c) <= prox
                mask := bitwise.or(mask, bitwise.lshift(1, et - 1))
    int cnt = 0
    for b = 0 to 9
        if bitwise.and(mask, bitwise.lshift(1, b)) != 0
            cnt += 1
    // Session window multiplier
    float dens = float(cnt)
    if sessBoost
        dens := dens * 1.25
    // High-magnet nearby boosts toward SINGULAR — only a magnet that actually supports
    // this direction (dir-matched, see f_nearHighMagnet fix) can raise the grade.
    [nearMag, magStr] = f_nearHighMagnet(c, 1.2, dir)
    int grade = math.round(dens)
    if nearMag and grade >= 3
        grade := math.min(5, grade + 1)
    if grade >= 5
        grade := 5
    // TRANSITION requires denser lattice
    if regime == "TRANSITION" and grade < 4 and dens < 3.5
        grade := math.min(grade, 2)
    grade

f_buildRisk(int dir, float zoneBot, float zoneTop, float sweepPx) =>
    float entry = c
    float sl = na
    if dir == 1
        float structSL = not na(sweepPx) ? sweepPx : not na(zoneBot) ? zoneBot : l
        sl := structSL - atr * stopATR * 0.35
    else
        float structSL = not na(sweepPx) ? sweepPx : not na(zoneTop) ? zoneTop : h
        sl := structSL + atr * stopATR * 0.35
    float risk = math.abs(entry - sl)
    // FIX #2: this used to floor `risk` for the TP math without touching `sl`, so a
    // tight structural stop (say 8 ticks) could get displayed next to a TP2 computed
    // off a much larger floored risk (say 20 ticks) — the labeled R-multiple didn't
    // match the actual stop distance. Now the SL itself is pushed out to match the
    // floor, so what's plotted and what's used for TP math always agree.
    if risk < atr * 0.5
        risk := atr * 0.5
        sl := dir == 1 ? entry - risk : entry + risk
    // TP1: VWAP if wrong side, else nearest opposing σ, else 1R
    float tp1 = na
    if dir == 1
        if entry < vwapVal
            tp1 := vwapVal
        else if entry < vwapUp1
            tp1 := vwapUp1
        else if entry < vwapUp2
            tp1 := vwapUp2
        else
            tp1 := entry + risk
    else
        if entry > vwapVal
            tp1 := vwapVal
        else if entry > vwapDn1
            tp1 := vwapDn1
        else if entry > vwapDn2
            tp1 := vwapDn2
        else
            tp1 := entry - risk
    float tp2 = dir == 1 ? entry + risk * rr : entry - risk * rr
    float magTP = f_nearestOpposingMagnet(dir, entry)
    float tp3 = not na(magTP) ? magTP : (dir == 1 ? entry + (tp2 - entry) * 1.5 : entry - (entry - tp2) * 1.5)
    [entry, sl, tp1, tp2, tp3, risk]
